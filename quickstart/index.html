Quickstart

Create your Stablecoin checkout page in 1 minute and start processing payments.

Create Checkout

POST
https://api.doura.fi/checkout/create-checkout

Required: token, amount, merchant_address, merchant_orderid, nonce
Optional: redirect_link, merchant_name, merchant_logo, merchant_email

merchant_address must be a Valid Solana Wallet Address with Corresponding Token Account.

See Create Checkout API Documentation for complete field specifications, and Error Responses for error handling.

Example Request

cURL
JavaScript
Python
curl --location --request POST 'https://api.doura.fi/checkout/create-checkout' \
--header 'Content-Type: application/json' \
--data-raw '{
    "token": "USDC",
    "amount": "5.53",
    "merchant_address": "AkskM1kf2QeDrD8WpqZAp1TRNuuKc6ouSZHP6DqXatdt",
    "merchant_orderid": "order_123456",
    "nonce": "'"$(openssl rand -base64 24)"'",
    "redirect_link": "https://example.com/success",
    "merchant_name": "Test Store",
    "merchant_logo": "http://localhost:3333/pipetil-icon.svg",
    "merchant_email": "test@teststore.com"
}'

var myHeaders = new Headers();
myHeaders.append("Content-Type", "application/json");

function randomBase64(bytes = 24) {
  const arr = new Uint8Array(bytes);
  crypto.getRandomValues(arr);
  return btoa(String.fromCharCode(...arr));
}

var raw = JSON.stringify({
    "token": "USDC",
    "amount": "5.53",
    "merchant_address": "AkskM1kf2QeDrD8WpqZAp1TRNuuKc6ouSZHP6DqXatdt",
    "merchant_orderid": "order_123456",
    "nonce": randomBase64(24),
    "redirect_link": "https://example.com/success",
    "merchant_name": "Test Store",
    "merchant_logo": "http://localhost:3333/pipetil-icon.svg",
    "merchant_email": "test@teststore.com"
});

var requestOptions = {
    method: "POST",
    headers: myHeaders,
    body: raw,
    redirect: "follow"
};

fetch("https://api.doura.fi/checkout/create-checkout", requestOptions)
    .then(response => response.text())
    .then(result => console.log(result))
    .catch(error => console.log("error", error));

import base64
import json
import requests
import secrets

url = "https://api.doura.fi/checkout/create-checkout"

payload = json.dumps({
   "token": "USDC",
   "amount": "5.53",
   "merchant_address": "AkskM1kf2QeDrD8WpqZAp1TRNuuKc6ouSZHP6DqXatdt",
   "merchant_orderid": "order_123456",
   "nonce": base64.b64encode(secrets.token_bytes(24)).decode(),
   "redirect_link": "https://example.com/success",
   "merchant_name": "Test Store",
   "merchant_logo": "http://localhost:3333/pipetil-icon.svg",
   "merchant_email": "test@teststore.com"
})
headers = {
   "Content-Type": "application/json",
}

response = requests.request("POST", url, headers=headers, data=payload)

print(response.text)

Always change nonce to a new value for each request, and do not reuse the same nonce for different requests.

Process Payment

Redirect to Checkout Page

Use doura_checkout_url to redirect payer to checkout page.

Redirect Back (Optional)

If redirect_link was provided, payer will be redirected after successful payment.

Query Payment Status

GET
https://api.doura.fi/checkout/status/{dtid}

Required: dtid (Doura Transaction ID)

See Get Status API Documentation for status queries, and Error Responses for error handling.

Example Request

cURL
JavaScript
Python
curl --location --request GET "https://api.doura.fi/checkout/status/f0884c1d4cd7978f703d8959cfb301cb"

fetch("https://api.doura.fi/checkout/status/f0884c1d4cd7978f703d8959cfb301cb")
   .then(response => response.text())
   .then(result => console.log(result))
   .catch(error => console.log('error', error));

import requests

url = "https://api.doura.fi/checkout/status/f0884c1d4cd7978f703d8959cfb301cb"

response = requests.request("GET", url)

print(response.text)
